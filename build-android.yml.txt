name: Build Android (APK for testing + AAB for Google Play)

on:
  workflow_dispatch:
    inputs:
      version_name:
        description: "Version name shown to users (e.g. 1.0.0)"
        default: "1.0.0"
      version_code:
        description: "Version code - must be higher than the previous upload (1, 2, 3 ...)"
        default: "1"

jobs:
  build:
    runs-on: ubuntu-latest
    timeout-minutes: 45
    env:
      VERSION_NAME: ${{ inputs.version_name }}
      VERSION_CODE: ${{ inputs.version_code }}
      NODE_ENV: production
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: 22

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 21

      - name: Install dependencies
        run: npm install --no-audit --no-fund --include=dev
        env:
          NODE_ENV: development

      - name: Unit tests (prayer times + notifications)
        run: npm test

      - name: Build web app
        run: npm run build

      - name: Make sure this is a production build
        run: |
          if grep -l "jsxDEV" dist/assets/*.js; then
            echo "Development build detected - aborting"; exit 1
          fi

      - name: Create Android project
        run: npx cap add android

      - name: Launcher icons + splash
        run: npx @capacitor/assets generate --android --iconBackgroundColor '#0A1328' --iconBackgroundColorDark '#0A1328' --splashBackgroundColor '#0A1328' --splashBackgroundColorDark '#0A1328'

      - name: Permissions, sounds, notification icon, version
        run: node scripts/android-setup.mjs

      - name: Sync web build into Android project
        run: npx cap sync android

      - name: Build debug APK + release AAB
        working-directory: android
        run: ./gradlew --no-daemon assembleDebug bundleRelease

      - name: Sign AAB with your upload key (only if secrets exist)
        env:
          KS_B64: ${{ secrets.ANDROID_KEYSTORE_BASE64 }}
          KS_PASS: ${{ secrets.ANDROID_KEYSTORE_PASSWORD }}
          KEY_ALIAS: ${{ secrets.ANDROID_KEY_ALIAS }}
          KEY_PASS: ${{ secrets.ANDROID_KEY_PASSWORD }}
        run: |
          AAB=android/app/build/outputs/bundle/release/app-release.aab
          if [ -z "$KS_B64" ]; then
            echo "::warning::No keystore secrets found - the AAB is UNSIGNED and Google Play will reject it."
            exit 0
          fi
          echo "$KS_B64" | base64 -d > upload.keystore
          jarsigner -keystore upload.keystore -storepass "$KS_PASS" -keypass "$KEY_PASS" \
            -sigalg SHA256withRSA -digestalg SHA-256 "$AAB" "$KEY_ALIAS"
          jarsigner -verify "$AAB" | tail -1
          rm -f upload.keystore

      - name: Upload files
        uses: actions/upload-artifact@v4
        with:
          name: almushaf-android-v${{ inputs.version_name }}-code${{ inputs.version_code }}
          path: |
            android/app/build/outputs/bundle/release/app-release.aab
            android/app/build/outputs/apk/debug/app-debug.apk
